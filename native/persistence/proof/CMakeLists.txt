add_library(flight_reconstruction_proof STATIC recipe.cpp)
target_include_directories(flight_reconstruction_proof PUBLIC include)
target_link_libraries(flight_reconstruction_proof PUBLIC flight::fdm_jsbsim)
add_executable(reconstruction_proof main.cpp)
target_link_libraries(reconstruction_proof PRIVATE flight_reconstruction_proof)
foreach(target flight_reconstruction_proof reconstruction_proof)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
if(BUILD_TESTING)
  find_program(FLIGHT_REPLAY_NODE NAMES node REQUIRED)
  add_test(NAME replay_reconstruction COMMAND "${FLIGHT_REPLAY_NODE}"
    "${CMAKE_SOURCE_DIR}/tests/replay/proof.mjs" "$<TARGET_FILE:reconstruction_proof>"
    "${CMAKE_BINARY_DIR}/replay-proof")
  set_tests_properties(replay_reconstruction PROPERTIES TIMEOUT 240 ENVIRONMENT "JSBSIM_DEBUG=0")
endif()
