add_library(flight_fdm_jsbsim STATIC src/session.cpp src/model.cpp src/protocol.cpp)
add_library(flight::fdm_jsbsim ALIAS flight_fdm_jsbsim)
target_compile_features(flight_fdm_jsbsim PUBLIC cxx_std_20)
target_include_directories(flight_fdm_jsbsim PUBLIC include)
target_include_directories(flight_fdm_jsbsim SYSTEM PRIVATE "${FLIGHT_DEPENDENCY_ROOT}/jsbsim/src")
target_link_libraries(flight_fdm_jsbsim PUBLIC flight::contracts PRIVATE flight::jsbsim)
if(UNIX)
  target_link_libraries(flight_fdm_jsbsim PRIVATE ${CMAKE_DL_LIBS})
endif()
# Content fingerprint is generated at build configure from actual first-party adapter inputs.
file(GLOB_RECURSE FDM_FINGERPRINT_INPUTS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*" "${CMAKE_CURRENT_SOURCE_DIR}/include/*")
list(SORT FDM_FINGERPRINT_INPUTS)
set(FDM_FINGERPRINT_TEXT "")
foreach(path IN LISTS FDM_FINGERPRINT_INPUTS)
  file(SHA256 "${path}" input_sha)
  file(RELATIVE_PATH relative_path "${CMAKE_CURRENT_SOURCE_DIR}" "${path}")
  string(APPEND FDM_FINGERPRINT_TEXT "${relative_path}:${input_sha}\n")
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${FDM_FINGERPRINT_INPUTS})
string(SHA256 FDM_SOURCE_FINGERPRINT "${FDM_FINGERPRINT_TEXT}")
target_compile_definitions(flight_fdm_jsbsim PRIVATE FLIGHT_FDM_COMPILER="${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION}" FLIGHT_FDM_SOURCE_FINGERPRINT="${FDM_SOURCE_FINGERPRINT}")
add_executable(fdm_harness src/main.cpp)
target_link_libraries(fdm_harness PRIVATE flight::fdm_jsbsim)
foreach(target flight_fdm_jsbsim fdm_harness)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
