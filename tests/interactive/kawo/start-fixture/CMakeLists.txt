cmake_minimum_required(VERSION 3.25)
project(PrivateKawoPhysicalStart LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This private observation packet supports Windows x64 only")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
# Explicit existing, separately qualified Root paths. No package download or native rebuild.
foreach(name FLIGHT_REPO_ROOT START_METADATA_HEADER)
  if(NOT DEFINED ${name} OR NOT IS_ABSOLUTE "${${name}}" OR NOT EXISTS "${${name}}")
    message(FATAL_ERROR "${name} must name a reviewed existing absolute input")
  endif()
endforeach()
add_subdirectory("${FLIGHT_REPO_ROOT}/native/sim_core/contracts" contracts)
add_subdirectory("${FLIGHT_REPO_ROOT}/native/world_core/ground" ground)
add_subdirectory("${FLIGHT_REPO_ROOT}/native/world_core/grid" grid)
add_library(start_fixture_sha STATIC "${FLIGHT_REPO_ROOT}/native/fdm_jsbsim/src/model.cpp")
target_include_directories(start_fixture_sha PUBLIC "${FLIGHT_REPO_ROOT}/native/fdm_jsbsim/include")
add_executable(kawo_start_author author.cpp)
target_link_libraries(kawo_start_author PRIVATE flight::resident_grid start_fixture_sha)
target_include_directories(kawo_start_author PRIVATE "${FLIGHT_REPO_ROOT}/native/world_core/grid/src")
set(fixture_targets start_fixture_sha kawo_start_author)
target_compile_definitions(kawo_start_author PRIVATE START_METADATA_HEADER="${START_METADATA_HEADER}")
# Author alone builds before a pose exists. Root freezes/reviews its output first,
# then supplies a separately authored reference header to enable the qualifier.
if(DEFINED START_REFERENCE_HEADER)
  foreach(name START_REFERENCE_HEADER START_JSBSIM_SOURCE START_JSBSIM_IMPLIB START_JSBSIM_DLL)
    if(NOT DEFINED ${name} OR NOT IS_ABSOLUTE "${${name}}" OR NOT EXISTS "${${name}}")
      message(FATAL_ERROR "${name} must name a reviewed existing absolute input")
    endif()
  endforeach()
  add_library(start_existing_jsbsim SHARED IMPORTED)
  set_target_properties(start_existing_jsbsim PROPERTIES IMPORTED_IMPLIB "${START_JSBSIM_IMPLIB}" IMPORTED_LOCATION "${START_JSBSIM_DLL}")
  add_executable(kawo_start_runic runic.cpp)
  target_include_directories(kawo_start_runic SYSTEM PRIVATE "${START_JSBSIM_SOURCE}/src")
  target_include_directories(kawo_start_runic PRIVATE "${FLIGHT_REPO_ROOT}/native/world_core/grid/src")
  target_link_libraries(kawo_start_runic PRIVATE flight::resident_grid start_fixture_sha start_existing_jsbsim)
  target_compile_definitions(kawo_start_runic PRIVATE START_METADATA_HEADER="${START_METADATA_HEADER}" START_REFERENCE_HEADER="${START_REFERENCE_HEADER}")
  list(APPEND fixture_targets kawo_start_runic)
  # Root copies only the reviewed existing DLL/dependencies into a fresh private
  # executable directory after reviewing identity/command closure. No copying to
  # the user package or implicit post-build DLL search/copy is performed here.
endif()
foreach(target IN LISTS fixture_targets)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /fp:strict /EHsc)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off)
  endif()
endforeach()
