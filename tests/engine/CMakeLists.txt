find_package(Python3 3.12 COMPONENTS Interpreter REQUIRED)
get_filename_component(PISTON_REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(piston_native_tests native.cpp)
add_executable(piston_fuel_mechanism native-mechanism.cpp)
foreach(target piston_native_tests piston_fuel_mechanism)
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_link_libraries(${target} PRIVATE flight::interactive flight::jsbsim)
  target_include_directories(${target} SYSTEM PRIVATE "${FLIGHT_DEPENDENCY_ROOT}/jsbsim/src")
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
add_test(NAME piston_engine_lifecycle COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/native-validate.py" --native "$<TARGET_FILE:piston_native_tests>" --mechanism "$<TARGET_FILE:piston_fuel_mechanism>" --model-root "${PISTON_REPOSITORY_ROOT}/native/fdm_jsbsim/models/original-piston-prop" --output "${CMAKE_BINARY_DIR}/piston-engine-proof")
set_tests_properties(piston_engine_lifecycle PROPERTIES ENVIRONMENT "JSBSIM_DEBUG=0" TIMEOUT 240)
