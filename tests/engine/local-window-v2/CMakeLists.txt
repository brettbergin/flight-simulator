# No substitute or automatically passing coupled test on the pristine route.
if(NOT FLIGHT_JSBSIM_HAS_EVENT_API)
  message(FATAL_ERROR "The piston coupled suite requires the reviewed event-aware backend option ON")
endif()
find_package(Python3 3.12 COMPONENTS Interpreter REQUIRED)
set(PISTON_REPOSITORY_ROOT "${CMAKE_SOURCE_DIR}")
if(NOT FLIGHT_JSBSIM_SOURCE_VARIANT STREQUAL "jsbsim-1.3.1-event-aware-coupled-midpoint-local-window-v2")
  message(FATAL_ERROR "Local-window suite requires its exact verified v2 source")
endif()
add_executable(piston_native_tests ../native.cpp)
add_executable(piston_fuel_mechanism ../native-mechanism.cpp)
set(PISTON_SUITE_METHOD event_aware_constant_power_v1)
if(FLIGHT_JSBSIM_HAS_COUPLED_API)
  set(PISTON_SUITE_METHOD event_aware_coupled_midpoint_v1)
endif()
foreach(target piston_native_tests piston_fuel_mechanism)
  if(FLIGHT_JSBSIM_HAS_COUPLED_API)
    target_compile_definitions(${target} PRIVATE FLIGHT_PISTON_TEST_COUPLED_MIDPOINT=1)
  endif()
  if(UNIX)
    target_link_libraries(${target} PRIVATE ${CMAKE_DL_LIBS})
  endif()
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_link_libraries(${target} PRIVATE flight::interactive flight::jsbsim)
  target_include_directories(${target} SYSTEM PRIVATE "${FLIGHT_JSBSIM_SOURCE_ROOT}/src")
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
add_test(NAME piston_engine_lifecycle COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../native-validate.py" --angular-method "${PISTON_SUITE_METHOD}" --native "$<TARGET_FILE:piston_native_tests>" --mechanism "$<TARGET_FILE:piston_fuel_mechanism>" --model-root "${PISTON_REPOSITORY_ROOT}/native/fdm_jsbsim/models/original-piston-prop" --output "${CMAKE_BINARY_DIR}/piston-engine-proof" --ratification "${FLIGHT_PISTON_EXECUTION_RATIFICATION}" --pretrial-ratification "${FLIGHT_PISTON_PRETRIAL_RATIFICATION}" --jsbsim-dll "$<TARGET_FILE:libJSBSim>")
set_tests_properties(piston_engine_lifecycle PROPERTIES ENVIRONMENT "JSBSIM_DEBUG=0" TIMEOUT 240)

if(FLIGHT_JSBSIM_HAS_COUPLED_API)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../shaft-method/probes-local-window-v2" "${CMAKE_CURRENT_BINARY_DIR}/coupled-probes")
  set(REFERENCE_REPOSITORY_ROOT "${CMAKE_SOURCE_DIR}" CACHE PATH "Frozen local-window reference checkout" FORCE)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../shaft-method/local-window-v2" "${CMAKE_CURRENT_BINARY_DIR}/local-window-references")
  # CMake test properties belong to the directory that registered the test.
  set_tests_properties(piston_engine_lifecycle PROPERTIES FIXTURES_REQUIRED coupled_native)
endif()
