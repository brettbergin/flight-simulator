cmake_minimum_required(VERSION 3.25)
project(FlightResidentGridQualification LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(FLIGHT_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_subdirectory("${FLIGHT_REPO_ROOT}/native/sim_core/contracts" contracts)
add_subdirectory("${FLIGHT_REPO_ROOT}/native/world_core/ground" ground)
add_subdirectory("${FLIGHT_REPO_ROOT}/native/world_core/grid" grid)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../contracts" ground_contract_tests)
find_package(Threads REQUIRED)
add_executable(resident_grid_tests native.cpp)
target_link_libraries(resident_grid_tests PRIVATE flight::resident_grid Threads::Threads)
add_executable(grid_payload_tests codec/native.cpp)
target_link_libraries(grid_payload_tests PRIVATE flight::resident_grid)
add_executable(grid_private_arithmetic_tests private_arithmetic.cpp)
target_link_libraries(grid_private_arithmetic_tests PRIVATE flight::resident_grid)
target_include_directories(grid_private_arithmetic_tests PRIVATE "${FLIGHT_REPO_ROOT}/native/world_core/grid/src")
foreach(target resident_grid_tests grid_payload_tests grid_private_arithmetic_tests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /fp:strict /EHsc)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off)
  endif()
endforeach()
add_test(NAME resident_grid COMMAND resident_grid_tests)
add_test(NAME grid_payload COMMAND grid_payload_tests)
add_test(NAME grid_private_arithmetic COMMAND grid_private_arithmetic_tests)
option(FLIGHT_GRID_RESOURCE_OBSERVER "Build the manual synthetic resource observer" OFF)
if(FLIGHT_GRID_RESOURCE_OBSERVER)
  add_executable(resident_grid_resource_observer resource-observer.cpp)
  target_link_libraries(resident_grid_resource_observer PRIVATE flight::resident_grid)
  if(WIN32)
    target_link_libraries(resident_grid_resource_observer PRIVATE Psapi)
  endif()
  if(MSVC)
    target_compile_options(resident_grid_resource_observer PRIVATE /W4 /WX /permissive- /fp:strict /EHsc)
  else()
    target_compile_options(resident_grid_resource_observer PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off)
  endif()
endif()
