cmake_minimum_required(VERSION 3.25)
project(FlightKawoPayloadQualification LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The native KAWO qualification supports x64 builds only.")
endif()
get_filename_component(FLIGHT_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
add_subdirectory("${FLIGHT_REPO_ROOT}/native/sim_core/contracts" contracts)
add_subdirectory("${FLIGHT_REPO_ROOT}/native/world_core/ground" ground)
add_subdirectory("${FLIGHT_REPO_ROOT}/native/world_core/grid" grid)

add_library(kawo_validation STATIC validation.cpp)
target_link_libraries(kawo_validation PUBLIC flight::resident_grid)
target_include_directories(kawo_validation PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(kawo_payload_validator main.cpp)
target_link_libraries(kawo_payload_validator PRIVATE kawo_validation)
add_executable(kawo_validation_tests validation_test.cpp)
target_link_libraries(kawo_validation_tests PRIVATE kawo_validation)
foreach(target kawo_validation kawo_payload_validator kawo_validation_tests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /fp:strict /EHsc)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off)
  endif()
endforeach()
add_test(NAME kawo_validation_refusals COMMAND kawo_validation_tests)
