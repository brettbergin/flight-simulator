cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(FlightContractTests LANGUAGES CXX)
  enable_testing()
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../native/sim_core/contracts" contracts)
endif()
add_executable(contract_boundary_tests contract_boundary_tests.cpp)
target_link_libraries(contract_boundary_tests PRIVATE flight::contracts)
if(MSVC)
  target_compile_options(contract_boundary_tests PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(contract_boundary_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME contract_boundary COMMAND contract_boundary_tests)
