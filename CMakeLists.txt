  string(JSON EXPECTED_SHA GET "${FLIGHT_LOCK}" sources ${INDEX} sha256)
  set(RECEIPT "${FLIGHT_DEPENDENCY_ROOT}/${ID}/.verified.json")
  if(NOT EXISTS "${RECEIPT}")
    message(FATAL_ERROR "Run tools/bootstrap/bootstrap.py before configure: missing verified ${ID}.")
  endif()

  string(JSON ACTUAL_SHA GET "${RECEIPT_CONTENT}" sha256)
  if(NOT ACTUAL_SHA STREQUAL EXPECTED_SHA)
    message(FATAL_ERROR "Dependency receipt does not match the current lock: ${ID}.")
  endif()
endforeach()

    foreach(target IN ITEMS flight_sqlite flight_local_players_core flight_local_players_windows flight_local_players_bridge)
      if(target STREQUAL "flight_sqlite")
        set(language C)
        set(source_directory "${PROJECT_SOURCE_DIR}")
        set(candidates "${FLIGHT_DEPENDENCY_ROOT}/sqlite/sqlite3.c")
      elseif(target STREQUAL "flight_local_players_core")
        set(language CXX)
        set(source_directory "${PROJECT_SOURCE_DIR}/native/persistence/local_players")
        set(candidates src/records.cpp src/codec.cpp src/store.cpp src/service.cpp src/storage_identity.cpp)
      elseif(target STREQUAL "flight_local_players_windows")
        set(language CXX)
        set(source_directory "${PROJECT_SOURCE_DIR}/native/persistence/local_players")
        set(candidates src/windows_filesystem.cpp)
      else()
        set(language CXX)
        set(source_directory "${PROJECT_SOURCE_DIR}/native/persistence/godot_bridge")
        set(candidates src/local_players.cpp src/register_types.cpp)
      endif()
      set(sources_json "[")
      set(source_separator "")
      foreach(source IN LISTS candidates)
        cmake_path(ABSOLUTE_PATH source BASE_DIRECTORY "${source_directory}" NORMALIZE OUTPUT_VARIABLE absolute_source)
        get_source_file_property(source_options "${absolute_source}" DIRECTORY "${source_directory}" COMPILE_OPTIONS)
        if(source_options STREQUAL "NOTFOUND")
          set(source_options "")
        endif()
        local_players_json_string(source_json "${absolute_source}")
        local_players_json_array(source_options_json ${source_options})
        string(APPEND sources_json "${source_separator}{\"path\":${source_json},\"options\":${source_options_json}}")
        set(source_separator ",")
      endforeach()
      string(APPEND sources_json "]")
      local_players_generated_array(definitions_json "$<TARGET_PROPERTY:${target},COMPILE_DEFINITIONS>")
      local_players_generated_array(includes_json "$<TARGET_PROPERTY:${target},INCLUDE_DIRECTORIES>")
      local_players_generated_array(options_json "$<TARGET_PROPERTY:${target},COMPILE_OPTIONS>")
      if(target STREQUAL "flight_local_players_bridge")
        local_players_generated_array(system_json "$<TARGET_PROPERTY:godot-cpp,INTERFACE_INCLUDE_DIRECTORIES>")
      else()
        set(system_json "[]")
      endif()
      get_target_property(export_define ${target} DEFINE_SYMBOL)
      if(export_define STREQUAL "export_define-NOTFOUND")
        if(target STREQUAL "flight_local_players_bridge")
          set(export_define "${target}_EXPORTS")
        else()
          set(export_define "")
        endif()
      endif()
      local_players_json_string(export_define_json "${export_define}")
      local_players_json_string(compiler_pdb_json "${CMAKE_BINARY_DIR}/local-players-compiler-pdb/${target}/${target}.pdb")
      string(APPEND local_players_targets_json "${target_separator}{\"target\":\"${target}\",\"language\":\"${language}\",\"defines\":${definitions_json},\"includes\":${includes_json},\"system_includes\":${system_json},\"options\":${options_json},\"export_define\":${export_define_json},\"msvc_runtime\":\"$<TARGET_GENEX_EVAL:${target},$<TARGET_PROPERTY:${target},MSVC_RUNTIME_LIBRARY>>\",\"cxx_modules\":\"OFF\",\"debug_information_format\":\"$<TARGET_GENEX_EVAL:${target},$<TARGET_PROPERTY:${target},MSVC_DEBUG_INFORMATION_FORMAT>>\",\"compiler_pdb\":${compiler_pdb_json},\"output\":\"$<TARGET_FILE:${target}>\",\"sources\":${sources_json}}")
      set(target_separator ",")
    endforeach()
    string(APPEND local_players_targets_json "]")
    local_players_json_string(standard_json "${CMAKE_CXX20_STANDARD_COMPILE_OPTION}")
    local_players_json_string(archive_rule_json "${CMAKE_CXX_CREATE_STATIC_LIBRARY}")
    local_players_json_string(link_rule_json "${CMAKE_CXX_CREATE_SHARED_LIBRARY}")
    local_players_json_string(linker_json "${CMAKE_LINKER}")
    local_players_json_string(librarian_json "${CMAKE_AR}")
    separate_arguments(shared_global_tokens WINDOWS_COMMAND "${CMAKE_SHARED_LINKER_FLAGS}")
    separate_arguments(shared_configuration_tokens WINDOWS_COMMAND "${CMAKE_SHARED_LINKER_FLAGS_${local_players_configuration_upper}}")
    local_players_json_array(shared_global_json ${shared_global_tokens})
    local_players_json_array(shared_configuration_json ${shared_configuration_tokens})
    local_players_generated_array(shared_target_json "$<TARGET_PROPERTY:flight_local_players_bridge,LINK_OPTIONS>")
    set(local_players_link_json "{\"schema\":\"LocalPlayersLinkPolicy/v1\",\"dll\":\"$<TARGET_FILE:flight_local_players_bridge>\",\"import_library\":\"$<TARGET_LINKER_FILE:flight_local_players_bridge>\",\"link_pdb\":\"$<TARGET_PDB_FILE:flight_local_players_bridge>\",\"map\":\"${CMAKE_BINARY_DIR}/local-players-storage-link.map\",\"global_flags\":${shared_global_json},\"configuration_flags\":${shared_configuration_json},\"target_options\":${shared_target_json},\"supervisor_policy\":\"cmake-3.31.8-vs_link_dll-nonincremental-no-manifest-inputs/v1\",\"manifest_inputs\":[],\"supervisor_appended_options\":[\"/MANIFEST:EMBED,ID=2\"]}")
    # Unique object paths/cwd are read from compile_commands file/directory/output
    # metadata only. Its original command String is evidence, never expected argv.
    file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/local-players-compile-policy-$<CONFIG>.json"
      CONTENT "{\"schema\":\"LocalPlayersCompilePolicy/v1\",\"rule_policy\":\"cmake-3.31.8-msvc-ninja-x64/v1\",\"configuration\":\"$<CONFIG>\",\"languages\":${local_players_language_json},\"targets\":${local_players_targets_json},\"cxx_standard_option\":${standard_json},\"linker\":${linker_json},\"librarian\":${librarian_json},\"archive_rule\":${archive_rule_json},\"shared_rule\":${link_rule_json},\"godot_cpp_archive\":\"$<TARGET_FILE:godot-cpp>\",\"link_policy\":${local_players_link_json}}\n")
  endif()
endif()
