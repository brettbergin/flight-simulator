cmake_minimum_required(VERSION 3.31)
cmake_policy(SET CMP0091 NEW)
project(FlightSimulator VERSION 0.0.1 LANGUAGES C CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL" CACHE STRING "Project ABI uses the dynamic MSVC CRT" FORCE)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_BUILD_RPATH_USE_ORIGIN ON)
if(MSVC)
  if(MSVC_VERSION LESS 1940 OR MSVC_VERSION GREATER_EQUAL 2000)
    message(FATAL_ERROR "Use an MSVC v143 compiler >=19.40; record the exact compiler patch.")
  endif()
  add_compile_options(/Zc:__cplusplus /utf-8)
elseif(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 13)
  message(FATAL_ERROR "Linux headless builds require GCC 13 or newer.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The current toolchain supports x64 builds only.")
endif()
set(FLIGHT_DEPENDENCY_ROOT "${CMAKE_SOURCE_DIR}/.local/deps" CACHE PATH "Checksum-verified dependency extraction")
option(FLIGHT_BUILD_GODOT_BINDINGS "Build the pinned standard-precision GDExtension bindings" ON)
file(READ "${CMAKE_SOURCE_DIR}/third_party/dependencies.lock.json" FLIGHT_LOCK)
string(JSON SOURCE_COUNT LENGTH "${FLIGHT_LOCK}" sources)
math(EXPR SOURCE_LAST "${SOURCE_COUNT} - 1")
foreach(INDEX RANGE 0 ${SOURCE_LAST})
  string(JSON ID GET "${FLIGHT_LOCK}" sources ${INDEX} id)
  string(JSON EXPECTED_SHA GET "${FLIGHT_LOCK}" sources ${INDEX} sha256)
  set(RECEIPT "${FLIGHT_DEPENDENCY_ROOT}/${ID}/.verified.json")
  if(NOT EXISTS "${RECEIPT}")
    message(FATAL_ERROR "Run tools/bootstrap/bootstrap.py before configure: missing verified ${ID}.")
  endif()
  file(READ "${RECEIPT}" RECEIPT_CONTENT)
  string(JSON ACTUAL_SHA GET "${RECEIPT_CONTENT}" sha256)
  if(NOT ACTUAL_SHA STREQUAL EXPECTED_SHA)
    message(FATAL_ERROR "Dependency receipt does not match the current lock: ${ID}.")
  endif()
endforeach()
# Upstream model/engine XML is not installed or redistributed by this build.
# Build a replaceable library only; aircraft fidelity is a later evidence gate.
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(SKBUILD ON)
set(BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(BUILD_PYTHON_MODULE OFF CACHE BOOL "" FORCE)
set(BUILD_JULIA_PACKAGE OFF CACHE BOOL "" FORCE)
set(SYSTEM_EXPAT OFF CACHE BOOL "" FORCE)
add_subdirectory("${FLIGHT_DEPENDENCY_ROOT}/jsbsim" "${CMAKE_BINARY_DIR}/deps/jsbsim" EXCLUDE_FROM_ALL)
get_target_property(JSBSIM_KIND libJSBSim TYPE)
if(NOT JSBSIM_KIND STREQUAL "SHARED_LIBRARY")
  message(FATAL_ERROR "JSBSim must remain a replaceable shared library.")
endif()
add_library(flight::jsbsim ALIAS libJSBSim)
add_library(flight_sqlite STATIC "${FLIGHT_DEPENDENCY_ROOT}/sqlite/sqlite3.c")
target_include_directories(flight_sqlite PUBLIC "${FLIGHT_DEPENDENCY_ROOT}/sqlite")
target_compile_definitions(flight_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION=1)
if(UNIX)
  target_link_libraries(flight_sqlite PUBLIC m pthread)
endif()
add_library(flight::sqlite ALIAS flight_sqlite)
if(FLIGHT_BUILD_GODOT_BINDINGS)
  set(GODOTCPP_USE_STATIC_CPP OFF CACHE BOOL "" FORCE)
  set(GODOTCPP_DISABLE_EXCEPTIONS OFF CACHE BOOL "" FORCE)
  set(GODOTCPP_PRECISION single CACHE STRING "" FORCE)
  set(GODOTCPP_TARGET template_release CACHE STRING "" FORCE)
  set(GODOTCPP_BUILD_PROFILE "${CMAKE_SOURCE_DIR}/tools/bootstrap/binding-profile.json" CACHE FILEPATH "" FORCE)
  add_subdirectory("${FLIGHT_DEPENDENCY_ROOT}/godot-cpp" "${CMAKE_BINARY_DIR}/deps/godot-cpp" EXCLUDE_FROM_ALL)
  set_target_properties(godot-cpp PROPERTIES SYSTEM ON PDB_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
endif()
add_executable(toolchain_smoke tools/bootstrap/smoke/main.cpp)
target_compile_features(toolchain_smoke PRIVATE cxx_std_20)
target_link_libraries(toolchain_smoke PRIVATE flight::jsbsim flight::sqlite)
# Treat upstream headers as external while retaining warnings-as-errors on project code.
target_include_directories(toolchain_smoke SYSTEM PRIVATE "${FLIGHT_DEPENDENCY_ROOT}/jsbsim/src")
if(MSVC)
  target_compile_options(toolchain_smoke PRIVATE /W4 /WX)
else()
  target_compile_options(toolchain_smoke PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(FLIGHT_BUILD_GODOT_BINDINGS)
  add_library(toolchain_binding_probe SHARED tools/bootstrap/smoke/binding_probe.cpp)
  target_link_libraries(toolchain_binding_probe PRIVATE godot::cpp)
  target_compile_features(toolchain_binding_probe PRIVATE cxx_std_20)
  if(MSVC)
    target_compile_options(toolchain_binding_probe PRIVATE /W4 /WX)
  else()
    target_compile_options(toolchain_binding_probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endif()
if(BUILD_TESTING)
  add_test(NAME native_toolchain_smoke COMMAND toolchain_smoke)
  set_tests_properties(native_toolchain_smoke PROPERTIES ENVIRONMENT "JSBSIM_DEBUG=0")
endif()
# Accepted contract module lands independently; no substitute/no-op contract tests.
if(EXISTS "${CMAKE_SOURCE_DIR}/native/sim_core/contracts/CMakeLists.txt")
  add_subdirectory(native/sim_core/contracts)
  if(BUILD_TESTING)
    if(NOT EXISTS "${CMAKE_SOURCE_DIR}/tests/contracts/CMakeLists.txt")
      message(FATAL_ERROR "Accepted contract module is missing its native test integration.")
    endif()
    add_subdirectory(tests/contracts)
  endif()
endif()
if(EXISTS "${CMAKE_SOURCE_DIR}/native/world_core/ground/CMakeLists.txt")
  add_subdirectory(native/world_core/ground)
  if(BUILD_TESTING)
    add_subdirectory(tests/ground)
  endif()
endif()
add_subdirectory(native/fdm_jsbsim)
if(BUILD_TESTING)
  add_subdirectory(tests/fdm)
endif()
if(FLIGHT_BUILD_GODOT_BINDINGS)
  add_subdirectory(native/godot_bridge)
endif()
if(EXISTS "${CMAKE_SOURCE_DIR}/native/persistence/proof/CMakeLists.txt")
  add_subdirectory(native/persistence/proof)
endif()
message(STATUS "Toolchain: ${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}; ${CMAKE_GENERATOR}; x64; C++20; shared JSBSim; dynamic MSVC CRT")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/toolchain-build-manifest.txt" CONTENT
"Project: FlightSimulator P1 toolchain proof
Compiler: ${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}
Compiler path: ${CMAKE_CXX_COMPILER}
CMake: ${CMAKE_VERSION}
Generator: ${CMAKE_GENERATOR}
Build type: $<CONFIG>
MSVC runtime: ${CMAKE_MSVC_RUNTIME_LIBRARY}
Dependencies: third_party/dependencies.lock.json
JSBSim: dynamic, source model assets excluded
Godot API: 4.5 single precision on pinned 4.7.2 engine; binding load is probed separately; exported simulator proof belongs to NATIVE-EXPORT
")
